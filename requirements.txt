# raises "concatenate() got an unexpected keyword argument 'casting'" inside
# functions.read_run().
#
#   python -m venv venv && source venv/bin/activate
#   pip install -r requirements.txt

scipy>=1.7

# ROOT I/O for the WCTE offline part-files (functions.read_run)
uproot>=5.0

# Parquet engine. Use pyarrow: fastparquet produced "Corrupted thrift data"
# errors when writing the large hit-level tables on Lustre.
pyarrow>=8.0
